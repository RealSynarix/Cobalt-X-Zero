#include "timer.h"
#include "pipeline.h"
#include <HardwareTimer.h>

static HardwareTimer *htim = nullptr;

static void timer_callback(void) {
  pipeline_tick();
}

void timer_init(void) {
  htim = new HardwareTimer(TIM3);
  htim->setOverflow(32000, HERTZ_FORMAT);
  htim->attachInterrupt(timer_callback);
  htim->setInterruptPriority(0, 0);
  htim->resume();
}

#include "pipeline.h"
#include <stm32g4xx.h>

#define DEBOUNCE_TICKS 96u
#define QUEUE_MASK 7u

static volatile uint8_t btn_state = 0;
static volatile uint16_t lock_lmb = 0;
static volatile uint16_t lock_rmb = 0;
static volatile uint16_t lock_mmb = 0;

static volatile uint8_t q_buf[QUEUE_MASK + 1u];
static volatile uint8_t q_head = 0;
static volatile uint8_t q_tail = 0;

static inline void q_push(uint8_t state) {
  uint8_t next = (q_head + 1u) & QUEUE_MASK;
  if (next != q_tail) {
    q_buf[q_head] = state;
    q_head = next;
  }
}

static inline void debounce(volatile uint16_t *lk, uint8_t idx, uint8_t pin) {
  if (*lk) {
    (*lk)--;
    return;
  }
  uint8_t pressed = !((GPIOB->IDR >> pin) & 1u);
  uint8_t was = (btn_state >> idx) & 1u;
  if (pressed == was) return;
  if (pressed) btn_state |= (1u << idx);
  else btn_state &= ~(1u << idx);
  *lk = DEBOUNCE_TICKS;
  q_push(btn_state);
}

void pipeline_init(void) {
  CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
  DWT->CYCCNT = 0;
  DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;

  RCC->AHB2ENR |= RCC_AHB2ENR_GPIOBEN;
  (void)RCC->AHB2ENR;
  GPIOB->MODER &= ~(0x3Fu << 6u);
  GPIOB->PUPDR = (GPIOB->PUPDR & ~(0x3Fu << 6u)) | (0x15u << 6u);
}

void pipeline_tick(void) {
  debounce(&lock_lmb, 0, 3);
  debounce(&lock_rmb, 1, 4);
  debounce(&lock_mmb, 2, 5);
}

uint8_t pipeline_dequeue(uint8_t *out) {
  if (q_head == q_tail) return 0;
  *out = q_buf[q_tail];
  q_tail = (q_tail + 1u) & QUEUE_MASK;
  return 1;
}

#include "usb_hid.h"
#include <Mouse.h>

extern "C" const uint8_t hid_report_descriptor[HID_REPORT_DESC_SIZE] = {
  0x05, 0x01, 0x09, 0x02, 0xA1, 0x01, 0x09, 0x01,
  0xA1, 0x00, 0x05, 0x09, 0x19, 0x01, 0x29, 0x05,
  0x15, 0x00, 0x25, 0x01, 0x95, 0x05, 0x75, 0x01,
  0x81, 0x02, 0x95, 0x01, 0x75, 0x03, 0x81, 0x03,
  0x05, 0x01, 0x09, 0x30, 0x09, 0x31, 0x09, 0x38,
  0x15, 0x81, 0x25, 0x7F, 0x75, 0x08, 0x95, 0x03,
  0x81, 0x06, 0xC0, 0xC0
};

void usb_hid_init(void) {
  Mouse.begin();
}

void usb_hid_send_buttons(uint8_t buttons) {
  static uint8_t prev = 0xFF;

  uint8_t changed = buttons ^ prev;
  if (!changed) return;
  prev = buttons;

  uint8_t falling = changed & ~buttons;
  uint8_t rising = changed & buttons;

  uint8_t rel_mask = 0;
  uint8_t prs_mask = 0;

  if (falling & 0x01) rel_mask |= MOUSE_LEFT;
  if (falling & 0x02) rel_mask |= MOUSE_RIGHT;
  if (falling & 0x04) rel_mask |= MOUSE_MIDDLE;
  if (rising & 0x01) prs_mask |= MOUSE_LEFT;
  if (rising & 0x02) prs_mask |= MOUSE_RIGHT;
  if (rising & 0x04) prs_mask |= MOUSE_MIDDLE;

  if (rel_mask) Mouse.release(rel_mask);
  if (prs_mask) Mouse.press(prs_mask);
}
